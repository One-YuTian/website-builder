## Overview

Global e-commerce shopping cart drawer and panel component built on Native Web Components. Integrates header item counter and close button, multi-item selected product list (with thumbnail skeletons, 2-line title skeletons, unit price, quantity stepper, and remove item), subtotal calculation, and checkout CTA buttons.

### Layout & Regions
- `header`: Header title skeleton, circular item count badge, and lightweight close button.
- `items`: Vertically scrollable item list; each row contains thumbnail skeleton, 2-line title, unit price, and stepper.
- `summary`: Footer subtotal summary row, primary checkout button, and secondary continue-shopping button.

### Dimensions & Responsiveness
- Follows global design system, drawer/panel max width 28rem (448px).
- Desktop: Functions as slide-out side drawer or embedded panel, aligned sticky to the right edge.
- Mobile (under 1024px): Full-width overlay drawer taking 100% viewport width.

### Attributes & Props
| Attribute | Description | Type | Default |
| --- | --- | --- | --- |
| `item-count` | Total number of selected items | `number` | `3` |
| `subtotal` | Formatted subtotal price string | `string` | `'$299.00'` |

### Events
- `cart-close`: Dispatched when clicking close button or backdrop.
- `cart-quantity-change`: Dispatched when stepper quantity changes.
- `cart-checkout`: Dispatched when clicking checkout button.

## Code

### HTML

```html
<cart></cart>
```

### JavaScript

```javascript
export class Cart extends HTMLElement {
  static observedAttributes = ['open', 'item-count', 'view-mode'];

  constructor() {
    super();
  }

  get open() { return this.hasAttribute('open'); }
  set open(val) { if (val) this.setAttribute('open', ''); else this.removeAttribute('open'); }

  get itemCount() { return Number(this.getAttribute('item-count')) || 3; }
  set itemCount(val) { this.setAttribute('item-count', String(val)); }

  get viewMode() { return this.getAttribute('view-mode') || 'inline'; }
  set viewMode(val) { this.setAttribute('view-mode', String(val)); }

  connectedCallback() {
    this.classList.add('block', 'w-full', 'min-w-0');
    this.render();
    this.attachEvents();
  }

  attachEvents() {
    this.addEventListener('click', (e) => {
      const target = e.target.closest('[data-cart-action]');
      if (!target) return;
      const action = target.getAttribute('data-cart-action');
      if (action === 'close') {
        this.open = false;
        this.dispatchEvent(new CustomEvent('cart-close', { bubbles: true, composed: true }));
      } else if (action === 'checkout') {
        this.dispatchEvent(new CustomEvent('cart-checkout', { bubbles: true, composed: true }));
      } else if (action === 'qty-minus' || action === 'qty-plus') {
        const delta = action === 'qty-plus' ? 1 : -1;
        this.dispatchEvent(new CustomEvent('cart-quantity-change', { detail: { delta }, bubbles: true, composed: true }));
      } else if (action === 'remove') {
        this.dispatchEvent(new CustomEvent('cart-item-remove', { bubbles: true, composed: true }));
      }
    });
  }

  render() {
    const isDrawer = this.viewMode === 'drawer';
    const items = Array.from({ length: Math.max(1, this.itemCount) });

    const innerContent = `
      <div class="h-full flex flex-col font-sans text-content bg-surface p-5 sm:p-6 rounded-2xl border border-outline/30 shadow-lg">
        <!-- Cart Header -->
        <div class="flex items-center justify-between pb-4 border-b border-outline/20 mb-4">
          <div class="flex items-center gap-2.5">
            <div class="h-5 w-28 bg-content/85 rounded-xs"></div>
            <span class="inline-flex items-center justify-center size-5 rounded-full bg-primary/10 text-primary text-xs font-semibold font-mono">
              ${this.itemCount}
            </span>
          </div>
          <button type="button" aria-label="关闭购物车" data-cart-action="close" class="size-8 rounded-full bg-accent/40 text-muted hover:text-content hover:bg-accent flex items-center justify-center transition-colors cursor-pointer">
            <span class="text-xl leading-none">&times;</span>
          </button>
        </div>

        <!-- Cart Items List -->
        <div class="flex-1 overflow-y-auto space-y-4 pr-1">
          ${items.map((_, idx) => `
            <div class="flex items-start gap-3.5 pb-4 border-b border-outline/15 last:border-b-0">
              <!-- Item Image Skeleton -->
              <div class="relative size-16 sm:size-20 shrink-0 bg-accent/40 rounded-lg flex items-center justify-center overflow-hidden border border-outline/20">
                <span class="relative block aspect-[6/5] w-1/2 overflow-hidden rounded-xs border border-muted">
                  <span class="absolute right-[15%] top-[15%] size-1/5 rounded-full bg-muted"></span>
                  <span class="absolute -bottom-1/4 left-[5%] size-3/4 rotate-45 border-l border-t border-muted"></span>
                  <span class="absolute -bottom-1/4 right-0 size-1/2 rotate-45 border-l border-t border-muted"></span>
                </span>
              </div>

              <!-- Item Details -->
              <div class="flex-1 min-w-0">
                <div class="h-3.5 w-11/12 bg-content/80 rounded-xs mb-1.5"></div>
                <div class="h-3 w-1/2 bg-muted/50 rounded-xs mb-2"></div>
                <div class="flex items-baseline gap-1.5 mb-2.5">
                  <span class="text-xs font-bold text-primary">$</span>
                  <div class="h-4 w-14 bg-primary/80 rounded-xs"></div>
                </div>

                <!-- Stepper & Actions -->
                <div class="flex items-center justify-between">
                  <div class="inline-flex items-center border border-outline/40 rounded-md bg-accent/20">
                    <button type="button" data-cart-action="qty-minus" class="size-6 flex items-center justify-center text-muted hover:text-content transition-colors cursor-pointer text-xs font-bold">-</button>
                    <span class="w-7 text-center font-mono text-xs font-medium text-content">1</span>
                    <button type="button" data-cart-action="qty-plus" class="size-6 flex items-center justify-center text-muted hover:text-content transition-colors cursor-pointer text-xs font-bold">+</button>
                  </div>
                  <button type="button" aria-label="删除商品" data-cart-action="remove" class="text-xs text-muted hover:text-red-600 transition-colors cursor-pointer p-1">
                    <span class="inline-block size-3.5 rounded-xs border border-muted/80"></span>
                  </button>
                </div>
              </div>
            </div>
          `).join('')}
        </div>

        <!-- Cart Footer -->
        <div class="border-t border-outline/20 pt-4 mt-2 space-y-3">
          <div class="flex items-center justify-between text-sm">
            <div class="h-4 w-12 bg-muted/60 rounded-xs"></div>
            <div class="flex items-baseline gap-1">
              <span class="text-sm font-bold text-primary">$</span>
              <div class="h-4.5 w-18 bg-primary rounded-xs"></div>
            </div>
          </div>
          <button type="button" data-cart-action="checkout" class="w-full py-3 px-4 rounded-xl bg-primary text-white font-medium shadow-sm hover:opacity-90 transition-opacity flex items-center justify-center cursor-pointer">
            <div class="h-3.5 w-24 bg-white/95 rounded-xs"></div>
          </button>
          <button type="button" data-cart-action="close" class="w-full py-2 px-4 rounded-lg border border-outline/40 text-xs text-muted hover:text-content hover:bg-accent/30 transition-colors flex items-center justify-center cursor-pointer">
            <div class="h-2.5 w-20 bg-muted/60 rounded-xs"></div>
          </button>
        </div>
      </div>
    `;

    if (isDrawer) {
      this.innerHTML = `
        <div class="fixed inset-0 z-50 ${this.open ? 'flex' : 'hidden'} justify-end bg-black/40 backdrop-blur-xs">
          <div class="w-full max-w-md h-full bg-surface shadow-2xl">
            ${innerContent}
          </div>
        </div>
      `;
    } else {
      this.innerHTML = innerContent;
    }
  }
}
```
