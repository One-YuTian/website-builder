## Overview

Shop catalog page, integrating page title, multi-dimensional filter sidebar, sorting toolbar, item counter, and responsive multi-column product card grid.

### Regions
Functional regions are annotated directly in the component code implementation via English comments.


### Dimensions & Responsiveness
- Follows Tailwind centered container（`w-full max-w-content mx-auto px-4 lg:px-6`） with PC max content width 1200px centered.
- Desktop (1024px+): 4 equal columns with optional 260px filter sidebar on left.
- Mobile (under 768px): 2 compact columns with wrapped toolbar.

### Attributes & Props
| Attribute | Description | Type | Default |
| --- | --- | --- | --- |
| `defaultSort` | Default sorting rule | `string` | `'default'` |
| `itemCount` | Initial product count | `number` | `8` |

### Events
- `product-navigate`: Dispatched when clicking product card.
- `shop-sort`: Dispatched when changing sorting order.

### JavaScript

```javascript
$(function() {
  const $grid = $('#products-grid');
  if (!$grid.length) return;

  const defaultProducts = [
    { id: 'prod-01', price: 120, rating: 5, discounted: true },
    { id: 'prod-02', price: 85, rating: 4, discounted: false },
    { id: 'prod-03', price: 210, rating: 5, discounted: true },
    { id: 'prod-04', price: 45, rating: 3, discounted: false },
    { id: 'prod-05', price: 160, rating: 4, discounted: true },
    { id: 'prod-06', price: 95, rating: 5, discounted: false },
    { id: 'prod-07', price: 320, rating: 5, discounted: true },
    { id: 'prod-08', price: 70, rating: 4, discounted: false }
  ];

  let products = [...defaultProducts];

  function render(list) {
    $grid.empty();
    $('#products-count').text(list.length);
    list.forEach(item => {
      const card = $(`<card-01 product-id="${item.id}" ${item.discounted ? 'discounted' : ''}></card-01>`);
      $grid.append(card);
    });
  }

  $('[data-dropdown="sort"] [data-dropdown-trigger]').on('click', function(e) {
    e.stopPropagation();
    const $menu = $(this).siblings('[data-dropdown-menu]');
    $('[data-dropdown-menu]').not($menu).addClass('hidden');
    $menu.toggleClass('hidden');
  });

  $('[data-dropdown="sort"] [data-dropdown-menu] [data-value]').on('click', function() {
    const val = $(this).data('value');
    $(this).closest('[data-dropdown-menu]').addClass('hidden');

    if (val === 'asc') {
      products.sort((a, b) => a.price - b.price);
    } else if (val === 'desc') {
      products.sort((a, b) => b.price - a.price);
    } else {
      products = [...defaultProducts];
    }
    render(products);
  });

  $(document).on('click', function(e) {
    if (!$(e.target).closest('[data-dropdown]').length) {
      $('[data-dropdown-menu]').addClass('hidden');
    }
  });

  render(products);
});
```
