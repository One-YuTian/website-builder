## Overview

Global sticky top navigation bar component, integrating brand LOGO skeleton, site-wide predictive search bar, user authentication & account portal, wishlist & cart counter badges, and horizontal category navigation menu. Features sticky positioning upon scroll and lightweight slide-out mobile drawer navigation.

### Regions
Functional regions are annotated directly in the component code implementation via English comments.


### Dimensions & Responsiveness
- Desktop (1024px+): Fixed height 68px~76px, follows global `content-container` with max content width 1200px centered, supports sticky positioning during page scrolling.
- Tablet & Mobile (under 1024px): Horizontal category list collapses, main bar retains Logo, search, and action icons; hamburger button triggers 320px smooth slide-out drawer.

### Attributes & Props
| Attribute | Description | Type | Default |
| --- | --- | --- | --- |
| `theme` | Color theme mode, supports dark and light inversion | `'dark' \| 'light'` | `'dark'` |
| `is-logged-in` | User authentication state, toggles account portal display | `boolean` | `false` |
| `cart-count` | Badge counter number on shopping cart icon | `number` | `0` |
| `wishlist-count` | Badge counter number on wishlist icon | `number` | `0` |

### Events
- `header-navigate`: Dispatched when clicking brand logo, nav links, or account menu items.
- `header-search`: Dispatched when submitting search form or pressing enter.
- `header-search-input`: Dispatched in real time as search input content changes for predictive suggestions.
- `header-cart`: Dispatched when clicking the cart icon to open drawer.

### JavaScript

```javascript
class Header extends HTMLElement {
  static observedAttributes = ['is-logged-in', 'cart-count', 'wishlist-count', 'theme'];

  constructor() {
    super();
    this._uid = 'header-06-' + Math.random().toString(36).substring(2, 8);
    this._searchSuggestions = [];
    window.$header = this;
  }

  get isLoggedIn() { return this.hasAttribute('is-logged-in'); }
  set isLoggedIn(val) { this.toggleAttribute('is-logged-in', Boolean(val)); }

  get cartCount() { return Number(this.getAttribute('cart-count')) || 0; }
  set cartCount(val) { this.setAttribute('cart-count', String(Math.max(0, Math.floor(Number(val) || 0)))); }

  get wishlistCount() { return Number(this.getAttribute('wishlist-count')) || 0; }
  set wishlistCount(val) { this.setAttribute('wishlist-count', String(Math.max(0, Math.floor(Number(val) || 0)))); }

  get theme() { return this.getAttribute('theme') || 'dark'; }
  set theme(val) { this.setAttribute('theme', String(val)); }

  get searchSuggestions() { return [...this._searchSuggestions]; }
  set searchSuggestions(list) {
    this._searchSuggestions = Array.isArray(list) ? list.slice(0, 6) : [];
    if (this.isConnected) this.updateSuggestions();
  }

  toggleMobileMenu(force) {
    const dialog = this.querySelector('dialog');
    if (!dialog) return;
    if (typeof force === 'boolean') {
      if (force && !dialog.open) dialog.showModal();
      else if (!force && dialog.open) dialog.close();
    } else {
      if (dialog.open) dialog.close();
      else dialog.showModal();
    }
  }

  toggleAccountDropdown() {
    const panel = this.querySelector('[data-region="account-panel"]');
    if (panel) panel.hidden = !panel.hidden;
  }

  updateCart(count) {
    this.cartCount = count;
  }

  updateWishlist(count) {
    this.wishlistCount = count;
  }

  connectedCallback() {
    this.classList.add('block', 'sticky', 'top-0', 'z-40');
    this.render();
    this.bindEvents();
  }

  disconnectedCallback() {
    this._controller?.abort();
    if (window.$header === this) delete window.$header;
  }

  attributeChangedCallback() {
    if (this.isConnected) this.render();
  }

  bindEvents() {
    this._controller?.abort();
    this._controller = new AbortController();
    const { signal } = this._controller;

    this.addEventListener('click', (e) => {
      const target = e.target;
      const actionEl = target.closest('[data-action]');
      if (!actionEl) return;

      const action = actionEl.dataset.action;
      if (action === 'menu') {
        this.toggleMobileMenu();
      } else if (action === 'close-menu') {
        this.toggleMobileMenu(false);
      } else if (action === 'account') {
        this.toggleAccountDropdown();
      } else if (action === 'cart') {
        this.emit('header-navigate', { destination: 'cart' });
      } else if (action === 'wishlist') {
        this.emit('header-navigate', { destination: 'wishlist' });
      } else if (action === 'login') {
        this.emit('header-navigate', { destination: 'login' });
      } else if (action === 'register') {
        this.emit('header-navigate', { destination: 'register' });
      } else if (action === 'nav') {
        const id = actionEl.dataset.navId;
        this.emit('header-navigate', { destination: 'category', id });
        this.toggleMobileMenu(false);
      }
    }, { signal });

    const input = this.querySelector('input[name="query"]');
    input?.addEventListener('input', (e) => {
      const query = e.target.value.trim();
      const placeholder = this.querySelector('[data-region="search-placeholder"]');
      if (placeholder) placeholder.hidden = Boolean(query);
      this.emit('header-search-input', { query });
    }, { signal });

    this.querySelector('form')?.addEventListener('submit', (e) => {
      e.preventDefault();
      const query = input?.value.trim() || '';
      if (query) {
        this.hideSuggestions();
        this.emit('header-search', { query });
      }
    }, { signal });

    document.addEventListener('pointerdown', (e) => {
      const panel = this.querySelector('[data-region="account-panel"]');
      const trigger = this.querySelector('[data-action="account"]');
      if (panel && !panel.contains(e.target) && !trigger?.contains(e.target)) {
        panel.hidden = true;
      }
      if (!this.querySelector('form')?.contains(e.target)) {
        this.hideSuggestions();
      }
    }, { signal });
  }

  hideSuggestions() {
    const list = this.querySelector('[data-region="search-suggestions"]');
    if (list) list.hidden = true;
  }

  updateSuggestions() {
    const list = this.querySelector('[data-region="search-suggestions"]');
    if (!list) return;
    list.replaceChildren();
    if (!this._searchSuggestions.length) {
      list.hidden = true;
      return;
    }
    list.hidden = false;
    this._searchSuggestions.forEach(item => {
      const li = document.createElement('li');
      li.className = 'p-2 hover:bg-accent/40 cursor-pointer rounded-lg flex items-center gap-3';
      li.innerHTML = `<span class="block size-10 rounded-sm bg-accent"></span><span class="flex-1 text-sm font-medium">${item.title || item.id}</span>`;
      li.addEventListener('click', () => {
        this.hideSuggestions();
        this.emit('header-navigate', { destination: 'product', id: item.id });
      });
      list.appendChild(li);
    });
  }

  emit(name, detail = {}) {
    this.dispatchEvent(new CustomEvent(name, { detail, bubbles: true, composed: true }));
  }

  render() {
    const isLight = this.theme === 'light';
    const focusRing = 'focus-visible:outline-2 focus-visible:outline-offset-4 focus-visible:outline-primary';
    const topBarCls = isLight ? 'bg-surface font-sans text-content border-b border-outline/30' : 'bg-primary font-sans text-white';
    const actionBtnCls = isLight ? 'hover:bg-accent/40 text-content' : 'hover:bg-white/10 text-white';
    const searchWrapCls = isLight ? 'bg-accent/25 border border-outline/40 text-content focus-within:ring-2 focus-within:ring-primary' : 'bg-surface text-content focus-within:ring-2 focus-within:ring-white';
    const subBarCls = isLight ? 'bg-surface/95 border-b border-outline/40 text-content' : 'bg-surface text-content shadow-xs border-b border-outline/20';

    const authSection = this.isLoggedIn
      ? `<div class="relative">
          <button type="button" data-action="account" class="flex size-9 items-center justify-center rounded-lg ${actionBtnCls} ${focusRing}">
            <span class="block size-6 rounded-full bg-accent"></span>
          </button>
          <div data-region="account-panel" hidden class="absolute right-0 top-full mt-2 w-48 rounded-xl border border-outline/60 bg-surface p-2 shadow-xl z-50 text-content">
            <button type="button" data-action="account-orders" class="w-full rounded-lg p-2 text-left hover:bg-accent/40 text-sm"><span class="block h-[0.75em] w-20 rounded-sm bg-accent"></span></button>
            <button type="button" data-action="account-wishlist" class="w-full rounded-lg p-2 text-left hover:bg-accent/40 text-sm"><span class="block h-[0.75em] w-16 rounded-sm bg-accent"></span></button>
            <button type="button" data-action="logout" class="w-full rounded-lg p-2 text-left hover:bg-accent/40 text-sm"><span class="block h-[0.75em] w-14 rounded-sm bg-accent"></span></button>
          </div>
        </div>`
      : `<div class="hidden sm:flex items-center gap-2">
          <button type="button" data-action="login" class="rounded-lg px-2.5 py-1 text-sm font-medium ${actionBtnCls} ${focusRing}"><span class="block h-[0.75em] w-10 rounded-sm bg-accent"></span></button>
          <span class="text-xs opacity-40">|</span>
          <button type="button" data-action="register" class="rounded-lg px-2.5 py-1 text-sm font-medium ${actionBtnCls} ${focusRing}"><span class="block h-[0.75em] w-12 rounded-sm bg-accent"></span></button>
        </div>`;

    this.innerHTML = `
      <header class="w-full">
        <div data-region="top-bar" class="${topBarCls} py-2.5">
          <div class="content-container flex items-center justify-between gap-4">
            <div data-region="logo">
              <div role="button" tabindex="0" data-action="nav" data-nav-id="home" class="flex items-center rounded-lg ${focusRing}">
                <span class="relative flex h-10 w-28 items-center justify-center overflow-hidden rounded-xl bg-accent"><span class="select-none text-xs font-semibold tracking-widest text-muted">LOGO</span></span>
              </div>
            </div>

            <div data-region="search-bar" class="hidden md:flex flex-1 max-w-md mx-4">
              <form class="relative w-full" role="search">
                <div class="relative flex items-center rounded-lg ${searchWrapCls} px-3 py-1.5 transition-shadow">
                  <span data-region="search-placeholder" class="pointer-events-none absolute left-3 select-none"><span class="block h-[0.75em] w-28 rounded-sm bg-accent"></span></span>
                  <input type="search" name="query" class="w-full bg-transparent text-sm outline-none placeholder:text-transparent" autocomplete="off" />
                  <button type="submit" class="flex size-7 shrink-0 items-center justify-center rounded-md ${actionBtnCls} ${focusRing}">
                    <span class="block size-3.5 rounded-full bg-accent"></span>
                  </button>
                </div>
                <ul data-region="search-suggestions" hidden class="absolute left-0 right-0 top-full mt-2 rounded-xl border border-outline/60 bg-surface p-2 shadow-xl z-50 text-content"></ul>
              </form>
            </div>

            <div class="flex items-center gap-1.5 sm:gap-3">
              <div data-region="auth">${authSection}</div>

              <button type="button" data-action="wishlist" class="relative flex size-9 items-center justify-center rounded-lg ${actionBtnCls} ${focusRing}">
                <span class="block size-4 rounded-sm bg-accent"></span>
                <span data-region="wishlist-badge" class="${this.wishlistCount > 0 ? '' : 'hidden'} absolute -right-1 -top-1 flex size-4 items-center justify-center rounded-full bg-surface text-primary text-[10px] font-bold shadow-xs">${this.wishlistCount}</span>
              </button>

              <button type="button" data-action="cart" class="relative flex size-9 items-center justify-center rounded-lg ${actionBtnCls} ${focusRing}">
                <span class="block size-4 rounded-sm bg-accent"></span>
                <span data-region="cart-badge" class="${this.cartCount > 0 ? '' : 'hidden'} absolute -right-1 -top-1 flex size-4 items-center justify-center rounded-full bg-surface text-primary text-[10px] font-bold shadow-xs">${this.cartCount}</span>
              </button>

              <button type="button" data-action="menu" class="flex size-9 items-center justify-center rounded-lg md:hidden ${actionBtnCls} ${focusRing}">
                <span class="flex flex-col gap-1"><span class="block h-0.5 w-4 bg-accent"></span><span class="block h-0.5 w-4 bg-accent"></span><span class="block h-0.5 w-4 bg-accent"></span></span>
              </button>
            </div>
          </div>
        </div>

        <div data-region="menu-bar" class="${subBarCls} py-2.5 hidden md:block">
          <div class="content-container">
            <nav class="flex items-center justify-center gap-8">
              <div role="button" tabindex="0" data-action="nav" data-nav-id="home" class="rounded-sm py-1 font-medium transition-colors hover:text-primary ${focusRing}"><span class="block h-[0.75em] w-12 rounded-sm bg-accent"></span></div>
              <div role="button" tabindex="0" data-action="nav" data-nav-id="shop" class="rounded-sm py-1 font-medium transition-colors hover:text-primary ${focusRing}"><span class="block h-[0.75em] w-10 rounded-sm bg-accent"></span></div>
              <div role="button" tabindex="0" data-action="nav" data-nav-id="about" class="rounded-sm py-1 font-medium transition-colors hover:text-primary ${focusRing}"><span class="block h-[0.75em] w-12 rounded-sm bg-accent"></span></div>
              <div role="button" tabindex="0" data-action="nav" data-nav-id="contact" class="rounded-sm py-1 font-medium transition-colors hover:text-primary ${focusRing}"><span class="block h-[0.75em] w-14 rounded-sm bg-accent"></span></div>
            </nav>
          </div>
        </div>

        <dialog data-region="mobile-drawer" class="fixed inset-0 m-0 h-full max-h-none w-72 max-w-[85vw] border-r border-outline/40 bg-surface p-0 shadow-2xl backdrop:bg-black/40">
          <div class="flex h-full flex-col p-4 text-content">
            <div class="flex items-center justify-between border-b border-outline/30 pb-4">
              <span class="relative flex h-8 w-24 items-center justify-center overflow-hidden rounded-lg bg-accent"><span class="select-none text-xs font-semibold tracking-widest text-muted">LOGO</span></span>
              <button type="button" data-action="close-menu" class="flex size-8 items-center justify-center rounded-lg hover:bg-accent/40 ${focusRing}"><span class="block size-4 rounded-full bg-accent"></span></button>
            </div>
            <nav class="flex flex-col gap-2 py-4">
              <div role="button" tabindex="0" data-action="nav" data-nav-id="home" class="rounded-lg p-2.5 font-medium hover:bg-accent/40"><span class="block h-[0.75em] w-16 rounded-sm bg-accent"></span></div>
              <div role="button" tabindex="0" data-action="nav" data-nav-id="shop" class="rounded-lg p-2.5 font-medium hover:bg-accent/40"><span class="block h-[0.75em] w-14 rounded-sm bg-accent"></span></div>
              <div role="button" tabindex="0" data-action="nav" data-nav-id="about" class="rounded-lg p-2.5 font-medium hover:bg-accent/40"><span class="block h-[0.75em] w-16 rounded-sm bg-accent"></span></div>
              <div role="button" tabindex="0" data-action="nav" data-nav-id="contact" class="rounded-lg p-2.5 font-medium hover:bg-accent/40"><span class="block h-[0.75em] w-20 rounded-sm bg-accent"></span></div>
            </nav>
            <div class="mt-auto border-t border-outline/30 pt-4 flex flex-col gap-2">
              <button type="button" data-action="login" class="w-full rounded-lg p-2 text-left hover:bg-accent/40 font-medium"><span class="block h-[0.75em] w-16 rounded-sm bg-accent"></span></button>
              <button type="button" data-action="register" class="w-full rounded-lg p-2 text-left hover:bg-accent/40 font-medium"><span class="block h-[0.75em] w-20 rounded-sm bg-accent"></span></button>
            </div>
          </div>
        </dialog>
      </header>`;
  }
}

if (!customElements.get('header-06')) {
```
